SELECT * FROM alunos WHERE idade BETWEEN 18 AND 20;

/*======================================== PARTE 8 ========================================*/

/*======================================== PARTE 10 ========================================*/

SELECT COUNT(*) FROM alunos;

SELECT COUNT(*) FROM alunos WHERE curso = 'Programação';

SELECT AVG(nota) FROM alunos;

SELECT MAX(nota) FROM alunos;

SELECT SUM(nota) FROM alunos;

SELECT MAX(nota) AS maior_nota, MIN(nota) AS menor_nota, AVG(nota) AS media FROM alunos;

/*======================================== PARTE 11 ========================================*/

SELECT cidade, COUNT(*) FROM alunos GROUP BY cidade;

SELECT curso, COUNT(*) FROM alunos GROUP BY curso;

SELECT cidade, AVG(nota) FROM alunos GROUP BY cidade;

SELECT curso, MAX(nota) FROM alunos GROUP BY curso;

/*======================================== PARTE 12 ========================================*/

SELECT cidade, COUNT(*) FROM alunos GROUP BY cidade HAVING COUNT(*) >= 3;

SELECT curso, COUNT(*) FROM alunos GROUP BY curso HAVING COUNT(*) > 3;

SELECT curso, AVG(nota) FROM alunos GROUP BY curso HAVING AVG(nota) > 7;

/*======================================== !DES FINAL! ========================================*/

SELECT cidade, COUNT(alunos) AS quantidade_alunos, AVG(nota) AS media FROM alunos 
GROUP BY cidade 
HAVING COUNT(alunos) >= 3 
AND AVG(nota) > 7 
ORDER BY AVG(nota) DESC;
