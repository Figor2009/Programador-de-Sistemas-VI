CREATE TABLE dimensao
(
	id SERIAL PRIMARY KEY,
	nome VARCHAR(4) NOT NULL
);

CREATE TABLE registro_de_vida
(
	id SERIAL PRIMARY KEY,
	nome VARCHAR(255) NOT NULL,
	omega NUMERIC NOT NULL,
	id_dimensao INT, FOREIGN KEY (id_dimensao) REFERENCES dimensao(id)
);

SELECT d.nome AS nome_dimensao, rv.nome, rv.omega, ROUND((rv.omega * 1.618), 3) AS Fator_N FROM registro_de_vida AS rv
INNER JOIN dimensao AS d
ON d.id = rv.id_dimensao
WHERE d.nome IN ('C875', 'C774')
AND(
rv.nome LIKE '%Douglas%' 
OR rv.nome LIKE '%Igor%'
OR rv.nome LIKE '%João%'
OR rv.nome LIKE '%Luiz%'
OR rv.nome LIKE '%Marcos%'
OR rv.nome LIKE '%Nicolas%'
)
ORDER BY rv.omega ASC;
